   106	def multipart(fields: dict, files: dict) -> tuple[bytes, str]:
   107	    bnd = "----drillrrboundary"
   108	    parts: list[bytes] = []
   109	    for k, v in fields.items():
   110	        parts.append(
   111	            f"--{bnd}\r\nContent-Disposition: form-data; name=\"{k}\"\r\n\r\n{v}\r\n".encode())
   112	    for k, (fn, blob, ct) in files.items():
   113	        parts.append(
   114	            f"--{bnd}\r\nContent-Disposition: form-data; name=\"{k}\"; filename=\"{fn}\"\r\n"
   115	            f"Content-Type: {ct}\r\n\r\n".encode() + blob + b"\r\n")
   116	    parts.append(f"--{bnd}--\r\n".encode())
   117	    return b"".join(parts), f"multipart/form-data; boundary={bnd}"

    89	def _enqueue(db: Session, *, op: str, target_key: str, payload: dict, upload_id) -> None:
    90	    existing = db.scalar(select(IndexTask).where(IndexTask.target_key == target_key))

   115	def canonical_record_key_sha(
   116	    title: str, document_date: str, url: str, origin_section: str = "breadth",
   117	) -> str:
   118	    """一.3 统一身份 + 三.2（2026-10-03）来源命名空间：
   119	
   120	    - breadth（含 GitHub feed 与基础索引全量）：sha256(json([t,d,u]))，
   121	      与历史公式【逐字节一致】——存量墓碑/遮蔽/替换高水位与基础索引
   122	      （sync_qa_index.key_digest）继续有效，不做迁移。
   123	    - industry/atsc：sha256(json([origin, t, d, u]))——独立数据源的身份
   124	      命名空间。历史缺陷：跨源同 title+date+url（空 URL 常见）会在
   125	      DB 与索引两侧互相覆盖/遮蔽/误删（一路墓碑屏蔽另一路全部 chunk）。
   126	      命名空间后，行业/ATSC 记录互不碰撞，也不与广度身份对齐（它们
   127	      本就不该跨源去重）。
   128	
   129	    与 canonical-qa server._record_key_sha_of 同源同值（origin 参数语义
   130	    一致；缺省 breadth 即历史公式，兼容所有旧调用）。
   131	    """
   132	    import hashlib as _hashlib
   133	    import json as _json
   134	    origin = (origin_section or "").strip().lower()
   135	    parts = ([origin] if origin in ("industry", "atsc") else []) + [
   136	        str(title or ""), str(document_date or ""), str(url or "")]
   137	    return _hashlib.sha256(_json.dumps(
   138	        parts, ensure_ascii=False).encode("utf-8")).hexdigest()
