   800	def reconcile_upload_storage() -> None:
   801	    db = SessionLocal()
   802	    try:
   803	        records = db.scalars(
   804	            select(UploadedFile).where(UploadedFile.checksum_sha256.is_(None))
   805	        ).all()
   806	        for record in records:
   807	            try:
   808	                path = safe_upload_path(record.stored_filename)
   809	            except ValueError:
   810	                record.storage_status = "invalid_path"
   811	                continue
   812	            if not path.is_file():
   813	                record.storage_status = "missing"
   814	                continue
   815	            digest = hashlib.sha256()
   816	            with open(path, "rb") as handle:
   817	                while chunk := handle.read(STREAM_CHUNK_BYTES):
   818	                    digest.update(chunk)
   819	            record.checksum_sha256 = digest.hexdigest()
   820	            record.storage_status = "stored"
   821	        db.commit()
   822	    finally:
   823	        db.close()
